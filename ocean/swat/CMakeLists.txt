cmake_minimum_required(VERSION 3.20)
project(swat_gold_element VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(SWAT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(SWAT_BOX3D_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../box3d" CACHE PATH "Box3D source checkout")
set(SWAT_RAYLIB_DIR "" CACHE PATH "Raylib installation (include/raylib.h, lib/libraylib.a)")
option(SWAT_BUILD_PLAYER "Build the Raylib player" ON)
option(SWAT_BUILD_TESTS "Build core and networking checks" ON)

set(BOX3D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX3D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX3D_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SWAT_BOX3D_DIR}" box3d)
add_subdirectory("${SWAT_ROOT}/vendor/enet" enet)
find_package(Threads REQUIRED)
add_library(swat_core STATIC body.c controller.c pose.c encounter.c weapons.c materials.c world.c mission.c equipment.c tactical.c overwatch.c generation.c acoustics.c audio_dsp.c sim.c)
add_executable(swat_layout_tool layout_tool.c)
target_link_libraries(swat_layout_tool PRIVATE swat_core)
target_include_directories(swat_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${SWAT_BOX3D_DIR}/include")
target_link_libraries(swat_core PUBLIC box3d Threads::Threads)
if(NOT WIN32)
    target_link_libraries(swat_core PUBLIC m)
endif()
add_library(swat_network STATIC protocol.c net.c replay.c)
target_include_directories(swat_network PUBLIC "${SWAT_ROOT}/vendor/enet/include")
target_link_libraries(swat_network PUBLIC swat_core enet)
add_library(swat_spatial STATIC spatial_audio.c)
target_include_directories(swat_spatial PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${SWAT_BOX3D_DIR}/include")
target_link_libraries(swat_spatial PRIVATE ${CMAKE_DL_LIBS})
add_executable(swat_audio_lab audio_lab.c)
target_link_libraries(swat_audio_lab PRIVATE swat_core swat_spatial)
if(WIN32)
    target_link_libraries(swat_network PUBLIC ws2_32 winmm)
endif()
if(MINGW)
    add_link_options(-static)
endif()
add_executable(swat_server server.c)
target_link_libraries(swat_server PRIVATE swat_network)
add_executable(swat_replay_tool replay_tool.c)
target_link_libraries(swat_replay_tool PRIVATE swat_network)
add_executable(swat_net_probe net_probe.c)
target_link_libraries(swat_net_probe PRIVATE swat_network)

if(SWAT_BUILD_PLAYER)
    find_package(raylib CONFIG QUIET)
    if(NOT TARGET raylib)
        find_path(SWAT_RAYLIB_INCLUDE raylib.h HINTS "${SWAT_RAYLIB_DIR}/include")
        find_library(SWAT_RAYLIB_LIBRARY NAMES raylib HINTS "${SWAT_RAYLIB_DIR}/lib")
        if(NOT SWAT_RAYLIB_INCLUDE OR NOT SWAT_RAYLIB_LIBRARY)
            message(FATAL_ERROR "Set SWAT_RAYLIB_DIR or SWAT_BUILD_PLAYER=OFF for a headless build")
        endif()
        add_library(raylib STATIC IMPORTED)
        set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${SWAT_RAYLIB_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${SWAT_RAYLIB_INCLUDE}")
    endif()
    add_executable(swat_player swat.c render.c frontend.c settings.c feedback.c sound_view.c)
    add_custom_target(swat_ui_assets
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:swat_player>/assets/ui"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${SWAT_ROOT}/resources/shared/Roboto-Regular.ttf" "$<TARGET_FILE_DIR:swat_player>/assets/ui/Roboto-Regular.ttf"
        COMMENT "Copying Gold Element HUD font" VERBATIM)
    add_dependencies(swat_player swat_ui_assets)
    target_include_directories(swat_player PRIVATE "${SWAT_ROOT}/src" "${SWAT_ROOT}/vendor")
    target_link_libraries(swat_player PRIVATE swat_network swat_spatial raylib)
    if(WIN32)
        target_link_libraries(swat_player PRIVATE opengl32 gdi32 winmm)
    elseif(APPLE)
        target_link_libraries(swat_player PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
    else()
        find_package(OpenGL REQUIRED)
        target_link_libraries(swat_player PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/config")
    configure_file("${SWAT_ROOT}/config/default.ini" "config/default.ini" COPYONLY)
    configure_file("${SWAT_ROOT}/config/swat.ini" "config/swat.ini" COPYONLY)
endif()

if(SWAT_BUILD_TESTS)
    enable_testing()
    foreach(check sim acoustics audio_dsp protocol net mission tactical overwatch generation spatial_audio foundation encounter)
        add_executable(swat_test_${check} tests/test_${check}.c)
        target_link_libraries(swat_test_${check} PRIVATE swat_network swat_spatial)
        if(MSVC)
            target_compile_options(swat_test_${check} PRIVATE /UNDEBUG)
        else()
            target_compile_options(swat_test_${check} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME swat_${check} COMMAND swat_test_${check} "${CMAKE_CURRENT_BINARY_DIR}/${check}-test.sgrp")
    endforeach()
    add_executable(swat_test_settings tests/test_settings.c settings.c)
    target_link_libraries(swat_test_settings PRIVATE swat_core)
    if(MSVC)
        target_compile_options(swat_test_settings PRIVATE /UNDEBUG)
    else()
        target_compile_options(swat_test_settings PRIVATE -UNDEBUG)
    endif()
    add_test(NAME swat_settings COMMAND swat_test_settings "${CMAKE_CURRENT_BINARY_DIR}/settings-test.ini")
endif()
