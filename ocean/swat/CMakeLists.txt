cmake_minimum_required(VERSION 3.20)
project(swat_gold_element VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(SWAT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(SWAT_BOX3D_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../box3d" CACHE PATH "Box3D source checkout")
set(SWAT_RAYLIB_DIR "" CACHE PATH "Raylib installation (include/raylib.h, lib/libraylib.a)")
option(SWAT_BUILD_PLAYER "Build the Raylib player" ON)
option(SWAT_BUILD_TESTS "Build core and networking checks" ON)
option(SWAT_BUILD_PHYSX_BENCHMARK "Build the optional CPU physics comparison (does not change the game backend)" OFF)
set(SWAT_PHYSX_DIR "" CACHE PATH "PhysX SDK source directory containing include/PxPhysicsAPI.h")
set(SWAT_PHYSX_LIB_DIR "" CACHE PATH "PhysX release static libraries directory")

# Presentation data importer stays independent of simulation and GPU libraries.
add_library(swat_character_asset STATIC character_asset.c)
target_include_directories(swat_character_asset PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
if(NOT WIN32)
    target_link_libraries(swat_character_asset PUBLIC m)
endif()
add_executable(swat_character_probe tests/character_probe.c)
target_link_libraries(swat_character_probe PRIVATE swat_character_asset)

set(BOX3D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX3D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX3D_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SWAT_BOX3D_DIR}" box3d)
add_subdirectory("${SWAT_ROOT}/vendor/enet" enet)
find_package(Threads REQUIRED)
add_library(swat_core STATIC body.c controller.c pose.c devices.c encounter.c locomotion.c weapons.c materials.c world.c motel.c storefront.c mission.c equipment.c tactical.c overwatch.c generation.c building.c acoustics.c audio_dsp.c sim.c)
add_executable(swat_layout_tool layout_tool.c)
target_link_libraries(swat_layout_tool PRIVATE swat_core)
if(NOT WIN32)
    find_package(OpenMP QUIET COMPONENTS C)
    add_executable(swat_training_benchmark training_benchmark.c protocol.c)
    target_link_libraries(swat_training_benchmark PRIVATE swat_core)
    if(OpenMP_C_FOUND)
        target_link_libraries(swat_core PUBLIC OpenMP::OpenMP_C)
        target_link_libraries(swat_training_benchmark PRIVATE OpenMP::OpenMP_C)
    endif()
endif()
target_include_directories(swat_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${SWAT_BOX3D_DIR}/include")
target_link_libraries(swat_core PUBLIC box3d Threads::Threads)
if(NOT WIN32)
    target_link_libraries(swat_core PUBLIC m)
endif()
if(SWAT_BUILD_PHYSX_BENCHMARK)
    if(NOT OpenMP_C_FOUND)
        message(FATAL_ERROR "The physics comparison requires OpenMP")
    endif()
    add_executable(swat_physics_benchmark physics_benchmark.cpp)
    target_compile_features(swat_physics_benchmark PRIVATE cxx_std_17)
    target_include_directories(swat_physics_benchmark PRIVATE "${SWAT_PHYSX_DIR}/include")
    target_compile_definitions(swat_physics_benchmark PRIVATE PX_PHYSX_STATIC_LIB)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(swat_physics_benchmark PRIVATE swat_core OpenMP::OpenMP_CXX)
    foreach(component PhysXExtensions PhysX PhysXCooking PhysXCommon PhysXPvdSDK PhysXFoundation)
        find_library(SWAT_${component}_LIB NAMES ${component}_static ${component}_static_64 PATHS "${SWAT_PHYSX_LIB_DIR}" NO_DEFAULT_PATH REQUIRED)
        target_link_libraries(swat_physics_benchmark PRIVATE "${SWAT_${component}_LIB}")
    endforeach()
    target_link_libraries(swat_physics_benchmark PRIVATE ${CMAKE_DL_LIBS})
endif()
add_library(swat_network STATIC protocol.c net.c replay.c)
target_include_directories(swat_network PUBLIC "${SWAT_ROOT}/vendor/enet/include")
target_link_libraries(swat_network PUBLIC swat_core enet)
add_library(swat_spatial STATIC spatial_audio.c)
target_include_directories(swat_spatial PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${SWAT_BOX3D_DIR}/include")
target_link_libraries(swat_spatial PRIVATE ${CMAKE_DL_LIBS})
add_executable(swat_audio_lab audio_lab.c)
target_link_libraries(swat_audio_lab PRIVATE swat_core swat_spatial)
if(WIN32)
    target_link_libraries(swat_network PUBLIC ws2_32 winmm)
endif()
if(MINGW)
    add_link_options(-static)
endif()
add_executable(swat_server server.c)
target_link_libraries(swat_server PRIVATE swat_network)
add_executable(swat_replay_tool replay_tool.c)
target_link_libraries(swat_replay_tool PRIVATE swat_network)
add_executable(swat_net_probe net_probe.c)
target_link_libraries(swat_net_probe PRIVATE swat_network)

if(SWAT_BUILD_PLAYER)
    find_package(raylib CONFIG QUIET)
    if(NOT TARGET raylib)
        find_path(SWAT_RAYLIB_INCLUDE raylib.h HINTS "${SWAT_RAYLIB_DIR}/include")
        find_library(SWAT_RAYLIB_LIBRARY NAMES raylib HINTS "${SWAT_RAYLIB_DIR}/lib")
        if(NOT SWAT_RAYLIB_INCLUDE OR NOT SWAT_RAYLIB_LIBRARY)
            message(FATAL_ERROR "Set SWAT_RAYLIB_DIR or SWAT_BUILD_PLAYER=OFF for a headless build")
        endif()
        add_library(raylib UNKNOWN IMPORTED)
        set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${SWAT_RAYLIB_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${SWAT_RAYLIB_INCLUDE}")
    endif()
    add_library(swat_character_render STATIC character_view.c character_runtime.c)
    target_link_libraries(swat_character_render PUBLIC swat_character_asset swat_core raylib)
    add_executable(swat_player swat.c render.c lighting.c weapon_art.c environment_art.c frontend.c settings.c feedback.c sound_view.c)
    add_custom_target(swat_ui_assets
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:swat_player>/assets/ui"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${SWAT_ROOT}/resources/shared/Roboto-Regular.ttf" "$<TARGET_FILE_DIR:swat_player>/assets/ui/Roboto-Regular.ttf"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_LIST_DIR}/assets" "$<TARGET_FILE_DIR:swat_player>/assets"
        COMMENT "Copying Gold Element runtime art" VERBATIM)
    add_dependencies(swat_player swat_ui_assets)
    target_include_directories(swat_player PRIVATE "${SWAT_ROOT}/src" "${SWAT_ROOT}/vendor")
    target_link_libraries(swat_player PRIVATE swat_network swat_spatial swat_character_render raylib)
    add_executable(swat_performance_tool performance_tool.c render.c lighting.c weapon_art.c environment_art.c sound_view.c)
    target_link_libraries(swat_performance_tool PRIVATE swat_network swat_spatial swat_character_render raylib)
    add_dependencies(swat_performance_tool swat_ui_assets)
    add_executable(swat_character_lab character_lab.c character_view.c lighting.c)
    target_link_libraries(swat_character_lab PRIVATE swat_character_asset swat_core raylib)
    if(WIN32)
        target_link_libraries(swat_character_lab PRIVATE opengl32 gdi32 winmm)
        target_link_libraries(swat_player PRIVATE opengl32 gdi32 winmm)
        target_link_libraries(swat_performance_tool PRIVATE opengl32 gdi32 winmm)
    elseif(APPLE)
        target_link_libraries(swat_character_lab PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
        target_link_libraries(swat_player PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
        target_link_libraries(swat_performance_tool PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
    else()
        find_package(OpenGL REQUIRED)
        target_link_libraries(swat_character_lab PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
        target_link_libraries(swat_player PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
        target_link_libraries(swat_performance_tool PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
    endif()
    if(SWAT_BUILD_TESTS)
        add_executable(swat_test_character_render tests/test_character_render.c lighting.c weapon_art.c environment_art.c)
        target_link_libraries(swat_test_character_render PRIVATE swat_character_render swat_network raylib)
        if(WIN32)
            target_link_libraries(swat_test_character_render PRIVATE opengl32 gdi32 winmm)
        elseif(APPLE)
            target_link_libraries(swat_test_character_render PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
        else()
            target_link_libraries(swat_test_character_render PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
        endif()
        if(MSVC)
            target_compile_options(swat_test_character_render PRIVATE /UNDEBUG)
        else()
            target_compile_options(swat_test_character_render PRIVATE -UNDEBUG)
        endif()
        add_executable(swat_test_environment_art tests/test_environment_art.c render.c lighting.c weapon_art.c environment_art.c)
        add_executable(swat_test_lighting tests/test_lighting.c lighting.c)
        add_executable(swat_test_weapon_art tests/test_weapon_art.c weapon_art.c lighting.c environment_art.c)
        target_link_libraries(swat_test_weapon_art PRIVATE swat_network swat_character_asset raylib)
        target_link_libraries(swat_test_lighting PRIVATE swat_network raylib)
        target_link_libraries(swat_test_environment_art PRIVATE swat_character_render swat_network raylib)
        if(WIN32)
            target_link_libraries(swat_test_environment_art PRIVATE swat_character_render opengl32 gdi32 winmm)
            target_link_libraries(swat_test_lighting PRIVATE opengl32 gdi32 winmm)
            target_link_libraries(swat_test_weapon_art PRIVATE opengl32 gdi32 winmm)
        elseif(APPLE)
            target_link_libraries(swat_test_environment_art PRIVATE swat_character_render "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
            target_link_libraries(swat_test_lighting PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
            target_link_libraries(swat_test_weapon_art PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
        else()
            target_link_libraries(swat_test_environment_art PRIVATE swat_character_render OpenGL::GL ${CMAKE_DL_LIBS})
            target_link_libraries(swat_test_lighting PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
            target_link_libraries(swat_test_weapon_art PRIVATE OpenGL::GL ${CMAKE_DL_LIBS})
        endif()
        if(MSVC)
            target_compile_options(swat_test_environment_art PRIVATE /UNDEBUG)
            target_compile_options(swat_test_lighting PRIVATE /UNDEBUG)
            target_compile_options(swat_test_weapon_art PRIVATE /UNDEBUG)
        else()
            target_compile_options(swat_test_environment_art PRIVATE -UNDEBUG)
            target_compile_options(swat_test_lighting PRIVATE -UNDEBUG)
            target_compile_options(swat_test_weapon_art PRIVATE -UNDEBUG)
        endif()
        add_dependencies(swat_test_environment_art swat_ui_assets)
        # Run explicitly with a display; ordinary headless CTest stays headless.
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/config")
    configure_file("${SWAT_ROOT}/config/default.ini" "config/default.ini" COPYONLY)
    configure_file("${SWAT_ROOT}/config/swat.ini" "config/swat.ini" COPYONLY)
endif()

if(SWAT_BUILD_TESTS)
    enable_testing()
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME swat_character_contract COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_LIST_DIR}/tests/test_character_contract.py" --probe $<TARGET_FILE:swat_character_probe>)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, scipy"
            RESULT_VARIABLE SWAT_NUMERICAL_ORACLE_AVAILABLE OUTPUT_QUIET ERROR_QUIET)
        if(SWAT_NUMERICAL_ORACLE_AVAILABLE EQUAL 0)
            add_test(NAME swat_character_numerical COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_LIST_DIR}/tests/test_character_asset.py" --probe $<TARGET_FILE:swat_character_probe>)
        endif()
    endif()
    if(UNIX)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME swat_launcher COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tests/test_launcher.py")
        endif()
    endif()
    foreach(check locomotion scenarios motel motel_nightstand motel_windows mounted_props storefront environment_props environment_binding sim acoustics audio_dsp protocol net mission tactical overwatch generation spatial_audio foundation encounter devices payloads impact_effects)
        add_executable(swat_test_${check} tests/test_${check}.c)
        target_link_libraries(swat_test_${check} PRIVATE swat_network swat_spatial)
        if(MSVC)
            target_compile_options(swat_test_${check} PRIVATE /UNDEBUG)
        else()
            target_compile_options(swat_test_${check} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME swat_${check} COMMAND swat_test_${check} "${CMAKE_CURRENT_BINARY_DIR}/${check}-test.sgrp")
    endforeach()
    add_executable(swat_test_settings tests/test_settings.c settings.c)
    target_link_libraries(swat_test_settings PRIVATE swat_core)
    if(MSVC)
        target_compile_options(swat_test_settings PRIVATE /UNDEBUG)
    else()
        target_compile_options(swat_test_settings PRIVATE -UNDEBUG)
    endif()
    add_test(NAME swat_settings COMMAND swat_test_settings "${CMAKE_CURRENT_BINARY_DIR}/settings-test.ini")
endif()
